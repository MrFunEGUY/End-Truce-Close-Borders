namespace = closeborders

# Find Truce Countries
country_event = {
	id = closeborders.1000
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		set_variable = { which = total_truce_countries value = 0 }

		every_country = {
			limit = {
				has_truce = root
			}
			set_country_flag = truce_country
			root = { change_variable = { which = total_truce_countries value = 1 } }
		}

		if = {
			inline_script = { script = effects/close_border_truce_effect NUM = 1 }
		}
		if = {
			inline_script = { script = effects/close_border_truce_effect NUM = 2 }
		}
		if = {
			inline_script = { script = effects/close_border_truce_effect NUM = 3 }
		}
		if = {
			inline_script = { script = effects/close_border_truce_effect NUM = 4 }
		}
		if = {
			inline_script = { script = effects/close_border_truce_effect NUM = 5 }
		}
		if = {
			inline_script = { script = effects/close_border_truce_effect NUM = 6 }
		}
		if = {
			inline_script = { script = effects/close_border_truce_effect NUM = 7 }
		}
		if = {
			inline_script = { script = effects/close_border_truce_effect NUM = 8 }
		}
		if = {
			inline_script = { script = effects/close_border_truce_effect NUM = 9 }
		}
		if = {
			inline_script = { script = effects/close_border_truce_effect NUM = 10 }
		}
	}

	after = {
		country_event = { id = closeborders.2000 }
	}
}

# CB 3001
country_event = { 
	id = closeborders.3001
	hide_window = yes
	is_triggered_only = yes

	immediate = { 
		end_truce = event_target:chosen_truce_country
		set_closed_borders = {
			who = event_target:chosen_truce_country
			status = yes
			forced = 0
		}
		event_target:chosen_truce_country = {
			set_closed_borders = {
				who = root
				status = yes
				forced = 0
			}
		}
	}
}
